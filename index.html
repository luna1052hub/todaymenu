let menuList = []; // 실시간 수집된 트렌드 메뉴 JSON 불러오기 async function loadTrendingMenuData() { try { const response = await fetch('./menu_data.json?v=' + new Date().getTime()); if (response.ok) { menuList = await response.json(); } else { throw new Error('파일 로드 실패'); } } catch (e) { // 백업 데이터를 사용하지 않고 빈 배열로 처리합니다. menuList = []; } } // 추천 버튼 클릭 이벤트 document.getElementById('recommendBtn').addEventListener('click', () => { const placeholder = document.getElementById('placeholder'); const menuGrid = document.getElementById('menuGrid'); // 데이터를 불러오지 못했거나 수집된 데이터가 없는 경우 if (!menuList || menuList.length === 0) { placeholder.style.display = 'block'; placeholder.innerHTML = '⚠️ 데이터를 불러오지 못했습니다.
잠시 후 다시 시도해 주세요.'; menuGrid.style.display = 'none'; return; } const selectedMenus = getRandomMenus(); menuGrid.innerHTML = ''; selectedMenus.forEach(item => { const card = document.createElement('div'); card.className = 'menu-card'; card.innerHTML = `
${item.emoji || '🔥'}
${item.name}
레시피 보기 👆
`; card.addEventListener('click', () => openRecipeModal(item)); menuGrid.appendChild(card); }); placeholder.style.display = 'none'; menuGrid.style.display = 'grid'; document.getElementById('recommendBtn').textContent = '🔄 다른 트렌드 메뉴 추천'; });
